cmake_minimum_required(VERSION 3.21)
project(QtSqliteWindows VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Sql)
qt_standard_project_setup()
qt_add_executable(QtSqliteWindows WIN32 src/main.cpp)
target_link_libraries(QtSqliteWindows PRIVATE Qt6::Widgets Qt6::Sql)
set_target_properties(QtSqliteWindows PROPERTIES WIN32_EXECUTABLE TRUE)
install(TARGETS QtSqliteWindows RUNTIME DESTINATION .)
